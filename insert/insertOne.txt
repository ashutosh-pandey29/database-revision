insertOne({}) = Inserts only one document into the collection; faster and safer when inserting single records.


db.student.insertOne({
  name:"rahul kumar",
  age:20,
  course:"BA",
  gender:"male"
});




===insert document with createdAt 
--------------------------------------

db.student.insertOne({

name:"parul",
  age:25,
  course:"MA",
  gender:"female",
  createdAt:new Date(),  
});

==Try inserting a document with a duplicate `_id`

db.student.insertOne({
_id:'6989bf9d89089961f3cebea5',
name:"sugam",
  age:25,
  course:"MA",
  gender:"female",
  createdAt:new Date(),  
});

==Insert a student with boolean field `isActive`
==Insert a document with an extra field not defined in schema

name:"raghav",
  age:28,
  course:"MA",
  gender:"male",
isActive:true,
address:{
  city:"pune",
}
});
